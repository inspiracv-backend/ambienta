# --- Autenticacion (ADR-006: Clerk) ------------------------------------------
# Validacion local del JWT de Clerk contra su JWKS publica. No se usa el SDK de
# Clerk a proposito: el SDK llama a la API de Clerk en cada request, mientras
# que validar la firma RS256 localmente no depende de que Clerk este arriba.
# Ver openspec/changes/integracion-clerk-auth/design.md §1.
python-jose[cryptography]==3.3.0
# Cliente HTTP para descargar la JWKS (se cachea en memoria, ver auth.py).
httpx==0.28.1
# Verificacion de firma de los webhooks de Clerk. Se usa la libreria del
# proveedor del protocolo en vez de reimplementar el HMAC a mano: es
# verificacion de firma, y ahi escribir criptografia propia es como se cuelan
# los bugs que nadie ve hasta que alguien los explota.
svix==1.42.0

# --- Almacenamiento de archivos (ADR-005: Backblaze B2) ----------------------
# B2 expone una API compatible con S3, asi que se usa el cliente de siempre en
# vez de la libreria propia de Backblaze: si manana el bucket se mueve a S3 o a
# R2, cambia el endpoint y no el codigo.
#
# `boto3` se importa **dentro de la funcion** que lo usa, no al tope del
# modulo: la API tiene que poder arrancar sin credenciales de almacenamiento.
boto3==1.35.71
